# The pipeline itself needs no third-party packages.
#
# scripts/run_pipeline.py uses only the standard library (argparse, pathlib,
# re, subprocess, time) and talks to PostgreSQL by invoking psql inside the
# Docker container, which the project already depends on. A database driver
# such as psycopg would add an install step for behaviour it does not need:
# the SQL files do the work, and the script only sequences them.
#
# One package is required, and only for generating the README figures:
matplotlib==3.11.2
#
# Verified on Python 3.15.2, the version this was built and run on.
